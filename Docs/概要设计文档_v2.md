RAG News Intelligence Platform 是一个基于检索增强生成（RAG）技术的智能新闻问答系统，通过自动化数据采集、智能语义检索和大语言模型问答，为用户提供精准、高效的新闻信息服务。

```python
from langchain_community.embeddings import HuggingFaceEmbeddings
from langchain_community.vectorstores import FAISS

class VectorService:
    def __init__(self):

        self.embeddings = HuggingFaceEmbeddings(
            model_name="sentence-transformers/all-MiniLM-L6-v2"
        )
        # 使用LangChain的FAISS VectorStore
        self.vectorstore = None
    
    def load_embedding_model(self)           # 初始化LangChain Embeddings
    def vectorize_text(self, text)           # 单文本向量化 (embeddings.embed_query)
    def batch_vectorize(self, texts)         # 批量向量化 (embeddings.embed_documents)
    def build_faiss_index(self, documents)   # 构建FAISS索引 (FAISS.from_documents)
```

```python
from langchain.retrievers import ContextualCompressionRetriever
from langchain.retrievers.document_compressors import CrossEncoderReranker
from langchain_community.cross_encoders import HuggingFaceCrossEncoder

class SearchService:
    def __init__(self):
        # 使用LangChain的Retriever接口
        self.base_retriever = None  # 从VectorStore创建
        # 使用LangChain的CrossEncoderReranker
        self.reranker = CrossEncoderReranker(
            model=HuggingFaceCrossEncoder(
                model_name="cross-encoder/ms-marco-MiniLM-L-6-v2"

        )
        # 使用LangChain的ContextualCompressionRetriever
        self.compression_retriever = None
```

```python
class WebCrawler:
    def __init__(self):
        self.session = requests.Session()
        self.session.headers.update({
            'User-Agent': 'Mozilla/5.0 (compatible; NewsBot/1.0)'
        })
    
    def crawl_website(self, url):
        try:
            response = self.session.get(url, timeout=10)
            soup = BeautifulSoup(response.content, 'html.parser')
            
            title = self.extract_title(soup)
            content = self.extract_content(soup)
            
            return {
                'title': title,
                'content': content,
                'source_url': url,
                'source_type': 'web'
            }
        except Exception as e:
            self.log_error(f"爬取失败: {url}, 错误: {str(e)}")
```

```python
        self.embedding_model = SentenceTransformer('all-MiniLM-L6-v2')
        self.rerank_model = SentenceTransformer('ms-marco-MiniLM-L-6-v2')
        self.faiss_index = None
        
    def vectorize_text(self, text):
        # 文本预处理
        processed_text = self.preprocess_text(text)
        # 生成向量
        vector = self.embedding_model.encode(processed_text)
        return vector
    
    def build_faiss_index(self, vectors):
        dimension = vectors.shape[1]
        index = faiss.IndexFlatIP(dimension)  # 内积相似度
        index.add(vectors.astype('float32'))
        self.faiss_index = index
        return index
    
    def search_similar(self, query_vector, top_k=10):
        scores, indices = self.faiss_index.search(
            query_vector.reshape(1, -1).astype('float32'), 
            top_k
        )
        return scores[0], indices[0]
```

#### 4.3.2 检索重排优化
```python
class RerankService:
    def __init__(self):
        self.rerank_model = SentenceTransformer('ms-marco-MiniLM-L-6-v2')
    
    def rerank_results(self, query, candidate_docs, top_k=5):
        # 计算query与每个候选文档的相关性分数
        pairs = [[query, doc['content']] for doc in candidate_docs]
        scores = self.rerank_model.predict(pairs)
        
        # 根据分数重新排序
        ranked_results = sorted(
            zip(candidate_docs, scores), 
            key=lambda x: x[1], 
            reverse=True
        )
        
        return ranked_results[:top_k]
```

### 4.4 RAG问答模块

#### 4.4.1 LLM集成设计
```python
class RAGService:
    def __init__(self):
        self.llm = Ollama(model="qwen3:8b")
        self.vector_service = VectorService()
        self.rerank_service = RerankService()
    
    def answer_question(self, query):
        # 1. 向量检索
        query_vector = self.vector_service.vectorize_text(query)
        candidates = self.vector_service.search_similar(query_vector, top_k=20)
        
        # 2. 重排优化
        reranked_docs = self.rerank_service.rerank_results(query, candidates)
        
        # 3. 构建提示词
        context = self.build_context(reranked_docs)
        prompt = self.build_prompt(query, context)
        
        # 4. LLM生成答案
        response = self.llm.generate(prompt)
        
        # 5. 后处理和格式化
        formatted_response = self.format_response(response, reranked_docs)
        
        return formatted_response
    
    def build_prompt(self, query, context):
        prompt_template = """
        基于以下新闻内容，回答用户的问题。请确保答案准确、客观，并标明信息来源。

        新闻内容：
        {context}

        问题：{query}

        请提供详细、准确的回答：
        """
        return prompt_template.format(context=context, query=query)
```

```python
class VectorManager:
    def __init__(self):
        self.index = self.load_or_create_index()
        self.id_mapping = self.load_id_mapping()
    
    def add_vectors(self, vectors, knowledge_ids):
        """添加向量到索引"""
        vector_ids = list(range(len(self.id_mapping), 
                               len(self.id_mapping) + len(vectors)))
        
        # 更新ID映射
        for i, knowledge_id in enumerate(knowledge_ids):
            self.id_mapping[vector_ids[i]] = knowledge_id
        
        # 添加到FAISS索引
        self.index.add(vectors.astype('float32'))
        
        # 保存更新
        self.save_index()
        self.save_id_mapping()
    
    def search_vectors(self, query_vector, top_k=20):
        """向量检索"""
        scores, vector_ids = self.index.search(
            query_vector.reshape(1, -1).astype('float32'), 
            top_k
        )
        
        # 转换为知识库ID
        knowledge_ids = [self.id_mapping[vid] for vid in vector_ids[0]]
        return scores[0], knowledge_ids
```

```python
import redis
import json

class CacheManager:
    def __init__(self):
        self.redis_client = redis.Redis(
            host='localhost', 
            port=6379, 
            db=0,
            decode_responses=True
        )
    
    def cache_search_results(self, query, results, expire_time=3600):
        """缓存搜索结果"""
        cache_key = f"search:{hashlib.md5(query.encode()).hexdigest()}"
        self.redis_client.setex(
            cache_key, 
            expire_time, 
            json.dumps(results)
        )
    
    def get_cached_results(self, query):
        """获取缓存的搜索结果"""
        cache_key = f"search:{hashlib.md5(query.encode()).hexdigest()}"
        cached_data = self.redis_client.get(cache_key)
        return json.loads(cached_data) if cached_data else None
```

| 数据库 | SQLite + FAISS | PostgreSQL + Milvus | 部署简单，满足需求 |
| 大模型 | Qwen3:8b | ChatGLM, Baichuan | 中文能力强，开源免费 |

### 11.3 关键技术决策

1. **选择FAISS而非Milvus**：FAISS部署简单，对于中小规模数据足够使用
2. **选择SQLite而非PostgreSQL**：开发阶段使用SQLite，生产环境可切换到PostgreSQL
3. **选择Ollama部署**：本地部署，数据安全，成本可控
4. **选择Vue 3 Composition API**：现代化开发体验，更好的TypeScript支持
5. **重新规划开发阶段**：解决依赖关系混乱，提高开发效率
